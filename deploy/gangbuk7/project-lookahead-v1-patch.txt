<script>
(function(){
function pc(){window.doc.projectControl=window.doc.projectControl||{history:[],issues:[],documents:[]};return window.doc.projectControl}
function days(a,b){if(!a||!b)return null;let x=new Date(a),y=new Date(b);if(isNaN(x)||isNaN(y))return null;return Math.round((y-x)/86400000)}
function stage(id){return (window.doc.stages||[]).find(s=>+s.id===+id)}
function openIssues(){return pc().issues.filter(x=>x.status!=='종결')}
function readyDocs(){let d=pc().documents;return {done:d.filter(x=>['완료','제출완료'].includes(x.status)).length,total:d.length}}
function forecastVariance(){let vals=(window.doc.stages||[]).map(s=>days(s.end,s.forecastEnd)).filter(v=>v!==null);return vals.length?Math.max(...vals):0}
function buildBoard(){let old=document.querySelector('.lookahead-board');if(old)old.remove();let anchor=document.querySelector('.gantt')||document.querySelector('.gantt-wrap')||document.querySelector('#gantt');if(!anchor)return;let issues=openIssues(),docs=pc().documents.filter(d=>!['완료','제출완료'].includes(d.status));let merge=[4,5,6].map(stage).filter(Boolean);let mergeDelay=Math.max(0,...merge.map(s=>days(s.end,s.forecastEnd)||0));let v=forecastVariance();let html='<section class="lookahead-board"><div class="lookahead-head"><b>PROJECT LOOK-AHEAD · 일정 영향 관제</b><span>30 / 60 / 90일 준비 · Critical Merge · Forecast</span></div><div class="lookahead-grid"><div class="la-col"><h5>30 DAYS · 즉시</h5>'+(docs.slice(0,3).map(d=>'<div class="la-item">제출자료 <strong>'+d.name+'</strong> · '+(d.status||'미착수')+'</div>').join('')||'<div class="la-empty">등록된 긴급 준비자료 없음</div>')+'</div><div class="la-col"><h5>60 DAYS · 선행준비</h5>'+(issues.slice(0,3).map(r=>'<div class="la-item"><strong>'+r.severity+'</strong> '+r.title+' · '+(r.owner||'담당 미정')+'</div>').join('')||'<div class="la-empty">등록된 OPEN 이슈 없음</div>')+'</div><div class="la-col"><h5>90 DAYS · NEXT GATE</h5><div class="la-item"><span class="gate-chip">02</span> 2027.01 소방 1차 심의 착수 기준</div><div class="la-item"><span class="gate-chip">MERGE</span> 04·05·06 완료 후 07 착수</div></div><div class="la-col"><h5>FORECAST · 일정영향</h5><div class="la-item">전체 최대 편차 <strong class="'+(v>0?'variance-plus':'variance-ok')+'">'+(v>0?'+'+v+'일':'지연 없음/미입력')+'</strong></div><div class="la-item">04·05·06 Merge <strong class="'+(mergeDelay>0?'variance-plus':'variance-ok')+'">'+(mergeDelay>0?'+'+mergeDelay+'일':'지연 없음/미입력')+'</strong></div></div></div></section>';anchor.insertAdjacentHTML('beforebegin',html)}
function addHistoryButton(){let bar=document.querySelector('.toolbar')||document.querySelector('.actions');if(!bar||document.getElementById('allHistoryBtn'))return;let b=document.createElement('button');b.id='allHistoryBtn';b.textContent='사업추진경위';b.className='btn';b.onclick=showHistory;bar.appendChild(b)}
function showHistory(){let hs=[...pc().history].sort((a,b)=>String(b.date).localeCompare(String(a.date)));let div=document.createElement('div');div.className='history-modal';div.innerHTML='<div class="history-card"><button class="history-close">닫기</button><h3>강북7구역 사업추진경위</h3><table class="history-all"><thead><tr><th>일자</th><th>단계</th><th>구분</th><th>추진내용</th><th>일정영향</th><th>담당</th></tr></thead><tbody>'+(hs.length?hs.map(h=>'<tr><td>'+String(h.date||'')+'</td><td>'+String(h.stageId||'')+'</td><td>'+String(h.type||'')+'</td><td>'+String(h.title||h.text||'')+'</td><td>'+String(h.impact||'-')+'</td><td>'+String(h.owner||'-')+'</td></tr>').join(''):'<tr><td colspan="6">등록된 실제 추진이력이 없습니다.</td></tr>')+'</tbody></table></div>';document.body.appendChild(div);div.querySelector('.history-close').onclick=()=>div.remove();div.onclick=e=>{if(e.target===div)div.remove()}}
function refresh(){try{buildBoard();addHistoryButton()}catch(e){console.warn('lookahead',e)}}
let r=window.renderAll;window.renderAll=function(){let x=r.apply(this,arguments);setTimeout(refresh,0);return x};
setTimeout(refresh,700);
})();
</script>
